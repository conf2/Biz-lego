const item = cat.items[itemIdx]; const w = item.w * UNIT, d = item.d * UNIT, h = item.h; const group = new THREE.Group(); const mat = new THREE.MeshPhysicalMaterial({ color: cat.color, roughness: 0.15, metalness: 0.05, clearcoat: 1.0, clearcoatRoughness: 0.05 });
